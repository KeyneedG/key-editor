cmake_minimum_required(VERSION 3.25)
project(tiny3d LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

option(TINY3D_HEADLESS "Build without the native window backend" OFF)
add_subdirectory(EryScript)
add_library(tiny3d STATIC src/renderer.cpp src/platform.cpp src/physics.cpp src/ui.cpp src/eryscript.cpp src/scene.cpp
    Rendering/Software/src/software.cpp)
target_link_libraries(tiny3d PUBLIC eryscript)
target_include_directories(tiny3d PUBLIC include)
target_compile_features(tiny3d PUBLIC cxx_std_26)
set_target_properties(tiny3d PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(tiny3d PRIVATE /W4 /permissive-)
else()
    target_compile_options(tiny3d PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(TINY3D_ENABLE_D3D11 "Build the optional Direct3D 11 rendering module" ON)
if(WIN32 AND NOT TINY3D_HEADLESS AND TINY3D_ENABLE_D3D11)
    add_subdirectory(Rendering/Direct3D11)
endif()

add_library(tiny3d_editor STATIC EXCLUDE_FROM_ALL Editor/src/editor.cpp Editor/src/panels.cpp Editor/src/dialogs.cpp)
target_link_libraries(tiny3d_editor PUBLIC tiny3d)
target_include_directories(tiny3d_editor PUBLIC Editor/include)
if(NOT MSVC)
    target_compile_options(tiny3d_editor PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TINY3D_HEADLESS)
    target_compile_definitions(tiny3d_editor PRIVATE TINY3D_HEADLESS)
endif()
if(WIN32 AND NOT TINY3D_HEADLESS)
    target_link_libraries(tiny3d_editor PRIVATE ole32 shell32 uuid)
endif()

option(TINY3D_BUILD_TESTS "Build scene/editor/UI integration checks" OFF)
if(TINY3D_BUILD_TESTS)
    enable_testing()
    if(TARGET tiny3d_d3d11)
        add_executable(tiny3d_d3d11_checks tests/d3d11_checks.cpp)
        target_link_libraries(tiny3d_d3d11_checks PRIVATE tiny3d_d3d11 tiny3d_editor)
        target_compile_definitions(tiny3d_d3d11_checks PRIVATE TINY3D_SOURCE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME d3d11_checks COMMAND tiny3d_d3d11_checks)
    endif()
    add_executable(tiny3d_editor_checks tests/editor_checks.cpp)
    target_link_libraries(tiny3d_editor_checks PRIVATE tiny3d_editor)
    add_test(NAME editor_checks COMMAND tiny3d_editor_checks)
    add_executable(tiny3d_scene_game_checks src/project_main.cpp)
    target_link_libraries(tiny3d_scene_game_checks PRIVATE tiny3d)
    target_compile_definitions(tiny3d_scene_game_checks PRIVATE TINY3D_SCENE_PROJECT
        TINY3D_PROJECT_MODE="game" TINY3D_PROJECT_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/scene-project-check")
    add_test(NAME scene_game_render COMMAND tiny3d_scene_game_checks --render "${CMAKE_CURRENT_BINARY_DIR}/scene-game.ppm")
    set_tests_properties(scene_game_render PROPERTIES DEPENDS editor_checks FAIL_REGULAR_EXPRESSION "EryScript.*error")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty-editor-check")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/empty-editor-check/engine.json" "{\"project\":\"\",\"mode\":\"editor\",\"autoStart\":false}\n")
    add_executable(tiny3d_empty_editor_checks src/project_main.cpp)
    target_link_libraries(tiny3d_empty_editor_checks PRIVATE tiny3d_editor)
    target_compile_definitions(tiny3d_empty_editor_checks PRIVATE TINY3D_EDITOR TINY3D_SCENE_PROJECT
        TINY3D_PROJECT_MODE="editor" TINY3D_PROJECT_DIRECTORY=""
        TINY3D_ENGINE_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/empty-editor-check"
        TINY3D_CONFIG_FILE="${CMAKE_CURRENT_BINARY_DIR}/empty-editor-check/engine.json")
    add_test(NAME empty_editor_render COMMAND tiny3d_empty_editor_checks --render "${CMAKE_CURRENT_BINARY_DIR}/empty-editor.ppm")
endif()
if(TINY3D_HEADLESS)
    target_compile_definitions(tiny3d PRIVATE TINY3D_HEADLESS)
endif()
if(WIN32 AND NOT TINY3D_HEADLESS)
    target_link_libraries(tiny3d PRIVATE user32 gdi32)
endif()

set(TINY3D_CONFIG "" CACHE FILEPATH "JSON file selecting a project and editor/game mode")
option(TINY3D_AUTO_START "Launch the selected project after building" ON)
if(TINY3D_CONFIG)
    include(project.cmake)
endif()
