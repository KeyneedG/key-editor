add_library(tiny3d_d3d11 STATIC src/d3d11.cpp)
target_link_libraries(tiny3d_d3d11 PUBLIC tiny3d PRIVATE d3d11 dxgi d3dcompiler)
target_include_directories(tiny3d_d3d11 PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(tiny3d_d3d11 INTERFACE TINY3D_HAS_D3D11)
if(MSVC)
    target_compile_options(tiny3d_d3d11 PRIVATE /W4 /permissive-)
else()
    target_compile_options(tiny3d_d3d11 PRIVATE -Wall -Wextra -Wpedantic)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/scene.hlsl" SCENE_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/present.hlsl" PRESENT_SHADER)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/scene.hlsl shaders/present.hlsl)
configure_file(src/shaders.hpp.in generated/shaders.hpp @ONLY)
