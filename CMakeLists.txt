cmake_minimum_required(VERSION 3.16)
project(tiny3d LANGUAGES CXX)

option(TINY3D_HEADLESS "Build without the native window backend" OFF)
add_library(tiny3d STATIC src/renderer.cpp src/platform.cpp)
target_include_directories(tiny3d PUBLIC include)
target_compile_features(tiny3d PUBLIC cxx_std_17)
set_target_properties(tiny3d PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(tiny3d PRIVATE /W4 /permissive-)
else()
    target_compile_options(tiny3d PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TINY3D_HEADLESS)
    target_compile_definitions(tiny3d PRIVATE TINY3D_HEADLESS)
endif()
if(WIN32 AND NOT TINY3D_HEADLESS)
    target_link_libraries(tiny3d PRIVATE user32 gdi32)
endif()

add_executable(demo examples/demo.cpp)
target_link_libraries(demo PRIVATE tiny3d)

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE tiny3d)
    add_test(NAME engine_tests COMMAND engine_tests)
    add_test(NAME render_demo COMMAND demo --render ${CMAKE_CURRENT_BINARY_DIR}/demo.ppm)
    if(WIN32 AND NOT TINY3D_HEADLESS)
        add_test(NAME window_smoke COMMAND demo --frames 3)
        set_tests_properties(window_smoke PROPERTIES TIMEOUT 10)
    endif()
endif()
